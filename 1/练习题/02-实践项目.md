```python
# calculator.py
import math
import json
from datetime import datetime

class Calculator:
    def __init__(self):
        self.history = []
    
    def add(self, a, b):
        """加法"""
        result = a + b
        self._save_history(f"{a} + {b} = {result}")
        return result
    
    def subtract(self, a, b):
        """减法"""
        result = a - b
        self._save_history(f"{a} - {b} = {result}")
        return result
    
    def multiply(self, a, b):
        """乘法"""
        result = a * b
        self._save_history(f"{a} * {b} = {result}")
        return result
    
    def divide(self, a, b):
        """除法"""
        if b == 0:
            raise ValueError("除数不能为零")
        result = a / b
        self._save_history(f"{a} / {b} = {result}")
        return result
    
    def power(self, a, b):
        """幂运算"""
        result = a ** b
        self._save_history(f"{a} ^ {b} = {result}")
        return result
    
    def square_root(self, a):
        """开方"""
        if a < 0:
            raise ValueError("负数不能开方")
        result = math.sqrt(a)
        self._save_history(f"√{a} = {result}")
        return result
    
    def _save_history(self, calculation):
        """保存计算历史"""
        self.history.append({
            "calculation": calculation,
            "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        })
    
    def show_history(self):
        """显示计算历史"""
        if not self.history:
            print("没有计算历史")
            return
        
        print("\n计算历史：")
        for i, record in enumerate(self.history, 1):
            print(f"{i}. {record['calculation']} ({record['timestamp']})")
    
    def clear_history(self):
        """清空历史记录"""
        self.history.clear()
        print("历史记录已清空")
```

```python
# file_manager.py
import os
import shutil
import glob
from datetime import datetime

class FileManager:
    def __init__(self, current_path="."):
        self.current_path = os.path.abspath(current_path)
    
    def get_current_path(self):
        """获取当前路径"""
        return self.current_path
    
    def change_directory(self, path):
        """切换目录"""
        if os.path.isdir(path):
            self.current_path = os.path.abspath(path)
            print(f"当前目录: {self.current_path}")
        else:
            print(f"目录不存在: {path}")
    
    def list_files(self, show_hidden=False):
        """列出文件"""
        files = os.listdir(self.current_path)
        if not show_hidden:
            files = [f for f in files if not f.startswith('.')]
        
        print(f"\n目录: {self.current_path}")
        print("-" * 50)
        for file in sorted(files):
            file_path = os.path.join(self.current_path, file)
            if os.path.isdir(file_path):
                print(f"[DIR]  {file}")
            else:
                size = os.path.getsize(file_path)
                print(f"[FILE] {file} ({size} bytes)")
    
    def create_file(self, filename):
        """创建文件"""
        file_path = os.path.join(self.current_path, filename)
        try:
            with open(file_path, 'w', encoding='utf-8') as file:
                file.write("")
            print(f"文件创建成功: {filename}")
        except Exception as e:
            print(f"创建文件失败: {e}")
    
    def create_directory(self, dirname):
        """创建目录"""
        dir_path = os.path.join(self.current_path, dirname)
        try:
            os.makedirs(dir_path, exist_ok=True)
            print(f"目录创建成功: {dirname}")
        except Exception as e:
            print(f"创建目录失败: {e}")
    
    def delete_file(self, filename):
        """删除文件"""
        file_path = os.path.join(self.current_path, filename)
        try:
            if os.path.isfile(file_path):
                os.remove(file_path)
                print(f"文件删除成功: {filename}")
            else:
                print(f"文件不存在: {filename}")
        except Exception as e:
            print(f"删除文件失败: {e}")
    
    def delete_directory(self, dirname):
        """删除目录"""
        dir_path = os.path.join(self.current_path, dirname)
        try:
            if os.path.isdir(dir_path):
                shutil.rmtree(dir_path)
                print(f"目录删除成功: {dirname}")
            else:
                print(f"目录不存在: {dirname}")
        except Exception as e:
            print(f"删除目录失败: {e}")
    
    def rename_file(self, old_name, new_name):
        """重命名文件"""
        old_path = os.path.join(self.current_path, old_name)
        new_path = os.path.join(self.current_path, new_name)
        try:
            os.rename(old_path, new_path)
            print(f"重命名成功: {old_name} -> {new_name}")
        except Exception as e:
            print(f"重命名失败: {e}")
    
    def copy_file(self, source, destination):
        """复制文件"""
        source_path = os.path.join(self.current_path, source)
        dest_path = os.path.join(self.current_path, destination)
        try:
            shutil.copy2(source_path, dest_path)
            print(f"复制成功: {source} -> {destination}")
        except Exception as e:
            print(f"复制失败: {e}")
    
    def move_file(self, source, destination):
        """移动文件"""
        source_path = os.path.join(self.current_path, source)
        dest_path = os.path.join(self.current_path, destination)
        try:
            shutil.move(source_path, dest_path)
            print(f"移动成功: {source} -> {destination}")
        except Exception as e:
            print(f"移动失败: {e}")
    
    def get_file_info(self, filename):
        """获取文件信息"""
        file_path = os.path.join(self.current_path, filename)
        try:
            if os.path.exists(file_path):
                stat = os.stat(file_path)
                print(f"\n文件信息: {filename}")
                print(f"大小: {stat.st_size} bytes")
                print(f"修改时间: {datetime.fromtimestamp(stat.st_mtime)}")
                print(f"访问时间: {datetime.fromtimestamp(stat.st_atime)}")
            else:
                print(f"文件不存在: {filename}")
        except Exception as e:
            print(f"获取文件信息失败: {e}")
    
    def search_files(self, pattern):
        """搜索文件"""
        search_path = os.path.join(self.current_path, pattern)
        files = glob.glob(search_path)
        if files:
            print(f"\n找到 {len(files)} 个文件:")
            for file in files:
                print(f"  {os.path.basename(file)}")
        else:
            print(f"没有找到匹配的文件: {pattern}")
```

```python
# models.py
class Student:
    def __init__(self, student_id, name, age, class_name):
        self.student_id = student_id
        self.name = name
        self.age = age
        self.class_name = class_name
        self.scores = {}
    
    def add_score(self, subject, score):
        """添加成绩"""
        self.scores[subject] = score
    
    def get_average_score(self):
        """计算平均分"""
        if not self.scores:
            return 0
        return sum(self.scores.values()) / len(self.scores)
    
    def to_dict(self):
        """转换为字典"""
        return {
            "student_id": self.student_id,
            "name": self.name,
            "age": self.age,
            "class_name": self.class_name,
            "scores": self.scores
        }
    
    @classmethod
    def from_dict(cls, data):
        """从字典创建对象"""
        student = cls(
            data["student_id"],
            data["name"],
            data["age"],
            data["class_name"]
        )
        student.scores = data.get("scores", {})
        return student
```

```python
# student_manager.py
import json
import csv
from models import Student

class StudentManager:
    def __init__(self):
        self.students = {}
        self.subjects = ["语文", "数学", "英语", "物理", "化学"]
    
    def add_student(self, student_id, name, age, class_name):
        """添加学生"""
        if student_id in self.students:
            print(f"学号 {student_id} 已存在")
            return False
        
        student = Student(student_id, name, age, class_name)
        self.students[student_id] = student
        print(f"学生 {name} 添加成功")
        return True
    
    def remove_student(self, student_id):
        """删除学生"""
        if student_id in self.students:
            student = self.students.pop(student_id)
            print(f"学生 {student.name} 删除成功")
            return True
        else:
            print(f"学号 {student_id} 不存在")
            return False
    
    def update_student(self, student_id, **kwargs):
        """更新学生信息"""
        if student_id not in self.students:
            print(f"学号 {student_id} 不存在")
            return False
        
        student = self.students[student_id]
        for key, value in kwargs.items():
            if hasattr(student, key):
                setattr(student, key, value)
        
        print(f"学生信息更新成功")
        return True
    
    def get_student(self, student_id):
        """获取学生信息"""
        return self.students.get(student_id)
    
    def list_students(self):
        """列出所有学生"""
        if not self.students:
            print("没有学生信息")
            return
        
        print("\n学生列表:")
        print("-" * 80)
        print(f"{'学号':<10} {'姓名':<10} {'年龄':<5} {'班级':<10} {'平均分':<8}")
        print("-" * 80)
        
        for student in self.students.values():
            avg_score = student.get_average_score()
            print(f"{student.student_id:<10} {student.name:<10} {student.age:<5} {student.class_name:<10} {avg_score:<8.2f}")
    
    def add_score(self, student_id, subject, score):
        """添加成绩"""
        if student_id not in self.students:
            print(f"学号 {student_id} 不存在")
            return False
        
        if subject not in self.subjects:
            print(f"科目 {subject} 不存在")
            return False
        
        if not (0 <= score <= 100):
            print("成绩必须在 0-100 之间")
            return False
        
        student = self.students[student_id]
        student.add_score(subject, score)
        print(f"成绩添加成功: {student.name} - {subject}: {score}")
        return True
    
    def get_rankings(self):
        """获取排名"""
        if not self.students:
            print("没有学生信息")
            return
        
        # 按平均分排序
        sorted_students = sorted(
            self.students.values(),
            key=lambda x: x.get_average_score(),
            reverse=True
        )
        
        print("\n成绩排名:")
        print("-" * 60)
        print(f"{'排名':<5} {'学号':<10} {'姓名':<10} {'平均分':<8}")
        print("-" * 60)
        
        for i, student in enumerate(sorted_students, 1):
            avg_score = student.get_average_score()
            print(f"{i:<5} {student.student_id:<10} {student.name:<10} {avg_score:<8.2f}")
    
    def get_statistics(self):
        """获取统计信息"""
        if not self.students:
            print("没有学生信息")
            return
        
        total_students = len(self.students)
        total_scores = [student.get_average_score() for student in self.students.values()]
        
        if total_scores:
            avg_score = sum(total_scores) / len(total_scores)
            max_score = max(total_scores)
            min_score = min(total_scores)
            pass_count = sum(1 for score in total_scores if score >= 60)
            pass_rate = pass_count / total_students * 100
            
            print("\n统计信息:")
            print(f"总学生数: {total_students}")
            print(f"平均分: {avg_score:.2f}")
            print(f"最高分: {max_score:.2f}")
            print(f"最低分: {min_score:.2f}")
            print(f"及格人数: {pass_count}")
            print(f"及格率: {pass_rate:.2f}%")
    
    def save_to_file(self, filename):
        """保存到文件"""
        data = {
            "students": {sid: student.to_dict() for sid, student in self.students.items()},
            "subjects": self.subjects
        }
        
        with open(filename, 'w', encoding='utf-8') as file:
            json.dump(data, file, ensure_ascii=False, indent=2)
        
        print(f"数据已保存到 {filename}")
    
    def load_from_file(self, filename):
        """从文件加载"""
        try:
            with open(filename, 'r', encoding='utf-8') as file:
                data = json.load(file)
            
            self.students = {}
            for sid, student_data in data["students"].items():
                self.students[sid] = Student.from_dict(student_data)
            
            self.subjects = data.get("subjects", self.subjects)
            print(f"数据已从 {filename} 加载")
        except FileNotFoundError:
            print(f"文件 {filename} 不存在")
        except Exception as e:
            print(f"加载文件失败: {e}")
    
    def export_to_csv(self, filename):
        """导出为 CSV"""
        with open(filename, 'w', newline='', encoding='utf-8') as file:
            writer = csv.writer(file)
            
            # 写入标题行
            headers = ["学号", "姓名", "年龄", "班级"] + self.subjects + ["平均分"]
            writer.writerow(headers)
            
            # 写入数据行
            for student in self.students.values():
                row = [
                    student.student_id,
                    student.name,
                    student.age,
                    student.class_name
                ]
                
                for subject in self.subjects:
                    row.append(student.scores.get(subject, ""))
                
                row.append(student.get_average_score())
                writer.writerow(row)
        
        print(f"数据已导出到 {filename}")
```

## 项目 4：文本分析工具

### 4.1 项目描述
创建一个文本分析工具，支持文本的统计分析、关键词提取、词频统计和文本比较等功能。

### 4.2 功能要求
- 文本统计：字符数、单词数、行数统计
- 词频分析：统计单词出现频率
- 关键词提取：提取重要关键词
- 文本比较：比较两个文本的差异
- 文件处理：支持多种文本格式

### 4.3 实现步骤

#### 步骤 1：创建文本分析器
```python
# text_analyzer.py
import re
from collections import Counter
import difflib

class TextAnalyzer:
    def __init__(self, text=""):
        self.text = text
    
    def load_from_file(self, filename):
        """从文件加载文本"""
        try:
            with open(filename, 'r', encoding='utf-8') as file:
                self.text = file.read()
            print(f"文本已从 {filename} 加载")
        except Exception as e:
            print(f"加载文件失败: {e}")
    
    def get_statistics(self):
        """获取文本统计信息"""
        if not self.text:
            print("没有文本内容")
            return
        
        # 基本统计
        char_count = len(self.text)
        word_count = len(self.text.split())
        line_count = len(self.text.splitlines())
        
        # 去除空行
        non_empty_lines = len([line for line in self.text.splitlines() if line.strip()])
        
        print("\n文本统计信息:")
        print(f"字符数: {char_count}")
        print(f"单词数: {word_count}")
        print(f"行数: {line_count}")
        print(f"非空行数: {non_empty_lines}")
    
    def get_word_frequency(self, top_n=10):
        """获取词频统计"""
        if not self.text:
            print("没有文本内容")
            return
        
        # 清理文本，只保留字母和空格
        clean_text = re.sub(r'[^a-zA-Z\s]', ' ', self.text.lower())
        words = clean_text.split()
        
        # 统计词频
        word_freq = Counter(words)
        
        print(f"\n词频统计 (前 {top_n} 个):")
        print("-" * 30)
        for word, count in word_freq.most_common(top_n):
            print(f"{word:<15} {count}")
    
    def get_keywords(self, top_n=10):
        """提取关键词"""
        if not self.text:
            print("没有文本内容")
            return
        
        # 简单的关键词提取（基于词频）
        clean_text = re.sub(r'[^a-zA-Z\s]', ' ', self.text.lower())
        words = clean_text.split()
        
        # 过滤停用词
        stop_words = {'the', 'a', 'an', 'and', 'or', 'but', 'in', 'on', 'at', 'to', 'for', 'of', 'with', 'by', 'is', 'are', 'was', 'were', 'be', 'been', 'have', 'has', 'had', 'do', 'does', 'did', 'will', 'would', 'could', 'should', 'may', 'might', 'must', 'can', 'this', 'that', 'these', 'those', 'i', 'you', 'he', 'she', 'it', 'we', 'they', 'me', 'him', 'her', 'us', 'them'}
        
        filtered_words = [word for word in words if word not in stop_words and len(word) > 2]
        
        # 统计词频
        word_freq = Counter(filtered_words)
        
        print(f"\n关键词 (前 {top_n} 个):")
        print("-" * 30)
        for word, count in word_freq.most_common(top_n):
            print(f"{word:<15} {count}")
    
    def compare_texts(self, other_text):
        """比较两个文本"""
        if not self.text or not other_text:
            print("文本内容为空")
            return
        
        # 使用 difflib 比较文本
        diff = difflib.unified_diff(
            self.text.splitlines(keepends=True),
            other_text.splitlines(keepends=True),
            fromfile='文本1',
            tofile='文本2'
        )
        
        print("\n文本比较结果:")
        print("-" * 50)
        for line in diff:
            print(line.rstrip())
    
    def search_text(self, pattern):
        """搜索文本"""
        if not self.text:
            print("没有文本内容")
            return
        
        # 使用正则表达式搜索
        matches = re.finditer(pattern, self.text, re.IGNORECASE)
        
        print(f"\n搜索 '{pattern}' 的结果:")
        print("-" * 50)
        
        count = 0
        for match in matches:
            count += 1
            start = max(0, match.start() - 20)
            end = min(len(self.text), match.end() + 20)
            context = self.text[start:end]
            print(f"{count}. ...{context}...")
        
        if count == 0:
            print("没有找到匹配的内容")
        else:
            print(f"\n总共找到 {count} 个匹配")
    
    def replace_text(self, old_text, new_text):
        """替换文本"""
        if not self.text:
            print("没有文本内容")
            return
        
        count = self.text.count(old_text)
        self.text = self.text.replace(old_text, new_text)
        print(f"替换完成，共替换 {count} 处")
    
    def save_to_file(self, filename):
        """保存到文件"""
        try:
            with open(filename, 'w', encoding='utf-8') as file:
                file.write(self.text)
            print(f"文本已保存到 {filename}")
        except Exception as e:
            print(f"保存文件失败: {e}")
```
