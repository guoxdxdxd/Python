# Python 数据类型详解

## 1. 数字类型 (Numbers)

### 1.1 整数 (int)
```python
# 基本整数
positive_int = 42
negative_int = -17
zero = 0

# 大整数（Python 自动处理）
big_int = 123456789012345678901234567890
print(big_int)

# 不同进制的整数
binary = 0b1010      # 二进制：10
octal = 0o755        # 八进制：493
hexadecimal = 0xFF   # 十六进制：255

print(f"二进制 0b1010 = {binary}")
print(f"八进制 0o755 = {octal}")
print(f"十六进制 0xFF = {hexadecimal}")

# 类型转换
str_to_int = int("123")
float_to_int = int(3.14)
print(f"字符串转整数：{str_to_int}")
print(f"浮点数转整数：{float_to_int}")
```

### 1.2 浮点数 (float)
```python
# 基本浮点数
pi = 3.14159
e = 2.71828
negative_float = -2.5

# 科学计数法
scientific = 1.23e4  # 12300.0
small_number = 1.23e-4  # 0.000123

print(f"科学计数法：{scientific}")
print(f"小数：{small_number}")

# 浮点数精度问题
a = 0.1 + 0.2
print(f"0.1 + 0.2 = {a}")  # 0.30000000000000004

# 使用 decimal 模块处理精确小数
from decimal import Decimal
precise_a = Decimal('0.1') + Decimal('0.2')
print(f"精确计算：{precise_a}")

# 类型转换
int_to_float = float(42)
str_to_float = float("3.14")
print(f"整数转浮点数：{int_to_float}")
print(f"字符串转浮点数：{str_to_float}")
```

### 1.3 复数 (complex)
```python
# 复数定义
complex1 = 3 + 4j
complex2 = complex(1, 2)  # 1 + 2j

print(f"复数1：{complex1}")
print(f"复数2：{complex2}")

# 复数运算
result = complex1 + complex2
print(f"复数相加：{result}")

# 获取实部和虚部
real_part = complex1.real
imag_part = complex1.imag
print(f"实部：{real_part}")
print(f"虚部：{imag_part}")

# 复数的模
magnitude = abs(complex1)
print(f"模长：{magnitude}")
```

## 2. 字符串类型 (str)

### 2.1 字符串创建
```python
# 单引号字符串
single_quote = 'Hello, World!'

# 双引号字符串
double_quote = "Hello, World!"

# 三引号字符串（多行）
multiline = """这是一个
多行字符串
可以包含换行符"""

# 原始字符串（不转义）
raw_string = r"C:\Users\Name\Documents"
print(raw_string)

# 字符串格式化
name = "张三"
age = 25
formatted = f"姓名：{name}，年龄：{age}"
print(formatted)
```

### 2.2 字符串操作
```python
text = "Hello, Python!"

# 字符串长度
print(f"长度：{len(text)}")

# 字符串索引
print(f"第一个字符：{text[0]}")
print(f"最后一个字符：{text[-1]}")

# 字符串切片
print(f"前5个字符：{text[:5]}")
print(f"后6个字符：{text[-6:]}")
print(f"中间部分：{text[2:8]}")

# 字符串连接
str1 = "Hello"
str2 = "World"
combined = str1 + " " + str2
print(f"连接结果：{combined}")

# 字符串重复
repeated = str1 * 3
print(f"重复结果：{repeated}")
```

### 2.3 字符串方法
```python
text = "  Hello, Python!  "

# 大小写转换
print(f"转大写：{text.upper()}")
print(f"转小写：{text.lower()}")
print(f"首字母大写：{text.title()}")
print(f"首字母大写：{text.capitalize()}")

# 去除空白
print(f"去除两端空白：'{text.strip()}'")
print(f"去除左端空白：'{text.lstrip()}'")
print(f"去除右端空白：'{text.rstrip()}'")

# 字符串查找
text = "Hello, Python!"
print(f"查找 'Python'：{text.find('Python')}")
print(f"查找 'Java'：{text.find('Java')}")  # 返回 -1
print(f"包含 'Python'：{'Python' in text}")

# 字符串替换
new_text = text.replace("Python", "Java")
print(f"替换结果：{new_text}")

# 字符串分割
text = "apple,banana,orange"
fruits = text.split(",")
print(f"分割结果：{fruits}")

# 字符串连接
fruits = ["apple", "banana", "orange"]
joined = ",".join(fruits)
print(f"连接结果：{joined}")
```

### 2.4 字符串格式化
```python
name = "张三"
age = 25
score = 85.5

# f-string 格式化（推荐）
formatted1 = f"姓名：{name}，年龄：{age}，分数：{score:.1f}"
print(formatted1)

# format() 方法
formatted2 = "姓名：{}，年龄：{}，分数：{:.1f}".format(name, age, score)
print(formatted2)

# % 格式化（旧式）
formatted3 = "姓名：%s，年龄：%d，分数：%.1f" % (name, age, score)
print(formatted3)

# 数字格式化
number = 1234567.89
print(f"千分位：{number:,}")
print(f"保留2位小数：{number:.2f}")
print(f"科学计数法：{number:.2e}")
print(f"百分比：{0.85:.1%}")
```

## 3. 布尔类型 (bool)

### 3.1 布尔值
```python
# 布尔值
true_value = True
false_value = False

print(f"真值：{true_value}")
print(f"假值：{false_value}")

# 布尔类型转换
print(f"数字转布尔：{bool(1)}")      # True
print(f"数字转布尔：{bool(0)}")      # False
print(f"字符串转布尔：{bool('hello')}")  # True
print(f"空字符串转布尔：{bool('')}")     # False
print(f"列表转布尔：{bool([1, 2, 3])}")  # True
print(f"空列表转布尔：{bool([])}")       # False
```

### 3.2 布尔运算
```python
a = True
b = False

# 逻辑与
print(f"True and False = {a and b}")
print(f"True and True = {a and a}")

# 逻辑或
print(f"True or False = {a or b}")
print(f"False or False = {b or b}")

# 逻辑非
print(f"not True = {not a}")
print(f"not False = {not b}")

# 短路求值
def func1():
    print("func1 被调用")
    return True

def func2():
    print("func2 被调用")
    return False

# 由于短路求值，func2 不会被调用
result = func1() or func2()
print(f"结果：{result}")
```

## 4. 列表类型 (list)

### 4.1 列表创建
```python
# 空列表
empty_list = []
empty_list2 = list()

# 包含元素的列表
numbers = [1, 2, 3, 4, 5]
fruits = ["苹果", "香蕉", "橙子"]
mixed = [1, "hello", 3.14, True]

# 使用 list() 构造函数
chars = list("hello")
print(f"字符串转列表：{chars}")

# 列表推导式
squares = [x**2 for x in range(10)]
print(f"平方数列表：{squares}")

# 嵌套列表
matrix = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
print(f"矩阵：{matrix}")
```

### 4.2 列表操作
```python
fruits = ["苹果", "香蕉", "橙子"]

# 访问元素
print(f"第一个元素：{fruits[0]}")
print(f"最后一个元素：{fruits[-1]}")

# 切片操作
print(f"前两个元素：{fruits[:2]}")
print(f"后两个元素：{fruits[-2:]}")
print(f"中间元素：{fruits[1:3]}")

# 修改元素
fruits[1] = "葡萄"
print(f"修改后：{fruits}")

# 添加元素
fruits.append("草莓")        # 在末尾添加
fruits.insert(1, "梨")      # 在指定位置插入
print(f"添加后：{fruits}")

# 删除元素
fruits.remove("葡萄")       # 删除指定值
del fruits[0]               # 删除指定索引
popped = fruits.pop()       # 删除并返回最后一个元素
print(f"删除后：{fruits}")
print(f"被删除的元素：{popped}")
```

### 4.3 列表方法
```python
numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 列表长度
print(f"长度：{len(numbers)}")

# 查找元素
print(f"元素 4 的索引：{numbers.index(4)}")
print(f"元素 1 的个数：{numbers.count(1)}")

# 排序
numbers.sort()              # 原地排序
print(f"排序后：{numbers}")

numbers.sort(reverse=True)  # 降序排序
print(f"降序排序：{numbers}")

# 反转
numbers.reverse()
print(f"反转后：{numbers}")

# 复制
numbers_copy = numbers.copy()
numbers_copy2 = numbers[:]  # 另一种复制方式
print(f"原列表：{numbers}")
print(f"复制列表：{numbers_copy}")
```

## 5. 元组类型 (tuple)

### 5.1 元组创建
```python
# 空元组
empty_tuple = ()
empty_tuple2 = tuple()

# 包含元素的元组
coordinates = (10, 20)
colors = ("红色", "绿色", "蓝色")
mixed = (1, "hello", 3.14, True)

# 单元素元组（注意逗号）
single = (42,)  # 必须有逗号
not_tuple = (42)  # 这不是元组，是整数

print(f"单元素元组：{single}")
print(f"类型：{type(single)}")
print(f"不是元组：{not_tuple}")
print(f"类型：{type(not_tuple)}")

# 元组解包
x, y = coordinates
print(f"x = {x}, y = {y}")

# 多重赋值
a, b, c = 1, 2, 3
print(f"a = {a}, b = {b}, c = {c}")
```

### 5.2 元组操作
```python
tuple1 = (1, 2, 3)
tuple2 = (4, 5, 6)

# 元组连接
combined = tuple1 + tuple2
print(f"连接结果：{combined}")

# 元组重复
repeated = tuple1 * 3
print(f"重复结果：{repeated}")

# 元组长度
print(f"长度：{len(tuple1)}")

# 查找元素
print(f"元素 2 的索引：{tuple1.index(2)}")
print(f"元素 1 的个数：{tuple1.count(1)}")

# 元组是不可变的
# tuple1[0] = 10  # 这会报错
```

## 6. 字典类型 (dict)

### 6.1 字典创建
```python
# 空字典
empty_dict = {}
empty_dict2 = dict()

# 包含键值对的字典
person = {
    "name": "张三",
    "age": 25,
    "city": "北京"
}

# 使用 dict() 构造函数
person2 = dict(name="李四", age=30, city="上海")

# 字典推导式
squares = {x: x**2 for x in range(5)}
print(f"平方字典：{squares}")

# 嵌套字典
nested = {
    "user1": {"name": "张三", "age": 25},
    "user2": {"name": "李四", "age": 30}
}
print(f"嵌套字典：{nested}")
```

### 6.2 字典操作
```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 访问值
print(f"姓名：{person['name']}")
print(f"年龄：{person.get('age')}")
print(f"邮箱：{person.get('email', '未设置')}")  # 提供默认值

# 修改值
person["age"] = 26
person["email"] = "zhang@example.com"
print(f"修改后：{person}")

# 删除键值对
del person["email"]
removed = person.pop("city", "默认值")
print(f"删除后：{person}")
print(f"被删除的值：{removed}")

# 字典长度
print(f"键值对数量：{len(person)}")

# 检查键是否存在
print(f"包含 'name' 键：{'name' in person}")
print(f"包含 'email' 键：{'email' in person}")
```

### 6.3 字典方法
```python
person = {"name": "张三", "age": 25, "city": "北京"}

# 获取所有键
keys = person.keys()
print(f"所有键：{list(keys)}")

# 获取所有值
values = person.values()
print(f"所有值：{list(values)}")

# 获取所有键值对
items = person.items()
print(f"所有键值对：{list(items)}")

# 遍历字典
for key in person:
    print(f"{key}: {person[key]}")

for key, value in person.items():
    print(f"{key}: {value}")

# 更新字典
new_info = {"age": 26, "email": "zhang@example.com"}
person.update(new_info)
print(f"更新后：{person}")

# 清空字典
person.clear()
print(f"清空后：{person}")
```

## 7. 集合类型 (set)

### 7.1 集合创建
```python
# 空集合
empty_set = set()  # 注意：{} 创建的是字典，不是集合

# 包含元素的集合
fruits = {"苹果", "香蕉", "橙子"}
numbers = {1, 2, 3, 4, 5}

# 从列表创建集合
list_to_set = set([1, 2, 2, 3, 3, 4])  # 自动去重
print(f"列表转集合：{list_to_set}")

# 集合推导式
squares = {x**2 for x in range(10)}
print(f"平方集合：{squares}")
```

### 7.2 集合操作
```python
set1 = {1, 2, 3, 4, 5}
set2 = {4, 5, 6, 7, 8}

# 并集
union = set1 | set2
union2 = set1.union(set2)
print(f"并集：{union}")

# 交集
intersection = set1 & set2
intersection2 = set1.intersection(set2)
print(f"交集：{intersection}")

# 差集
difference = set1 - set2
difference2 = set1.difference(set2)
print(f"差集：{difference}")

# 对称差集
symmetric_diff = set1 ^ set2
symmetric_diff2 = set1.symmetric_difference(set2)
print(f"对称差集：{symmetric_diff}")

# 添加元素
set1.add(6)
set1.update([7, 8, 9])
print(f"添加后：{set1}")

# 删除元素
set1.remove(9)      # 如果元素不存在会报错
set1.discard(10)    # 如果元素不存在不会报错
popped = set1.pop() # 删除并返回任意元素
print(f"删除后：{set1}")
print(f"被删除的元素：{popped}")
```

## 8. 类型检查和转换

### 8.1 类型检查
```python
# 检查变量类型
value = 42
print(f"类型：{type(value)}")
print(f"是整数：{isinstance(value, int)}")
print(f"是数字：{isinstance(value, (int, float))}")

# 检查多个类型
text = "hello"
print(f"是字符串或列表：{isinstance(text, (str, list))}")
```

### 8.2 类型转换
```python
# 数字转换
str_num = "123"
int_num = int(str_num)
float_num = float(str_num)

print(f"字符串转整数：{int_num}")
print(f"字符串转浮点数：{float_num}")

# 字符串转换
num = 42
text = str(num)
print(f"数字转字符串：{text}")

# 列表转换
text = "hello"
char_list = list(text)
print(f"字符串转列表：{char_list}")

# 集合转换
num_list = [1, 2, 2, 3, 3, 4]
num_set = set(num_list)
print(f"列表转集合：{num_set}")

# 元组转换
num_tuple = tuple(num_list)
print(f"列表转元组：{num_tuple}")
```
